<script>
const videos = [
    { title: "Class 1: Intro", id: "OnBVrIpL94Q" },
    { title: "Class 2: Exam", id: "Y8sXdbQ7034" },
    { title: "Class 3: Management", id: "4NcwzS_j_M0" },
    { title: "Class 4: Advanced", id: "h5S6Yg7ApL4" },
    { title: "Class 5: Summary", id: "iMmLGdt_Jp0" }
];

function togglePlaylist() {
    const list = document.getElementById('video-list');
    const arrow = document.getElementById('p-arrow');
    const isActive = list.classList.toggle('active');
    arrow.innerHTML = isActive ? '▲' : '▼';
}

const listContainer = document.getElementById('video-list');
videos.forEach(v => {
    const b = document.createElement('button');
    b.className = 'playlist-btn';
    b.innerHTML = `▶ ${v.title}`;
    b.onclick = () => openVideo(v.id);
    listContainer.appendChild(b);
});

let player = null;
let wmInterval = null;

function openVideo(id) {
    const plyrContainer = document.getElementById('dynamic-plyr');
    plyrContainer.innerHTML = `<div class="js-player" data-plyr-provider="youtube" data-plyr-embed-id="${id}"></div>`;
    
    document.getElementById('popup').style.display = 'block';

    player = new Plyr('.js-player', {
        autoplay: true,
        controls: ['play-large', 'play', 'progress', 'current-time', 'mute', 'volume', 'fullscreen'],
        youtube: { noCookie: true, rel: 0, modestbranding: 1, iv_load_policy: 3 }
    });

    // Start with a forced blackout until the video actually plays
    player.on('ready', () => {
        const pElement = document.querySelector('.plyr');
        pElement.classList.add('force-blackout');
        
        // Move watermark inside the Plyr container so it shows in fullscreen
        const wm = document.getElementById('watermark');
        pElement.appendChild(wm);
    });

    player.on('playing', () => {
        document.querySelector('.plyr').classList.remove('force-blackout');
    });

    player.on('buffering', () => {
        // Plyr adds .plyr--buffering automatically, which triggers our CSS
    });

    startWatermarkLogic();
}

function startWatermarkLogic() {
    const wm = document.getElementById('watermark');
    if (wmInterval) clearInterval(wmInterval);
    const move = () => {
        wm.style.left = (Math.random() * 60 + 5) + '%';
        wm.style.top = (Math.random() * 60 + 10) + '%';
    };
    move();
    wmInterval = setInterval(move, 3000);
}

function closePopup() {
    document.getElementById('popup').style.display = 'none';
    if (player) player.destroy();
    if (wmInterval) clearInterval(wmInterval);
}
